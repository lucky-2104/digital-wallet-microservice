  user-db:
  	image: postgres:15
  	container_name: user_db
  	environment:
  	  POSTGRES_DB: user_db
  	  POSTGRES_USER: postgre
  	  POSTGRES_PASSWORD: root
  	ports:
  	  - "5433:5432"
  
  wallet-db:
    image: postgres:15
    container_name: wallet_db
    environment:
      POSTGRES_DB: wallet_db
      POSTGRES_USER: postgre

    ports:
      - "5434:5432"

    container_name: eureka-server
    ports:
      - "8761:8761"

    container_name: user-service

    depends_on:
